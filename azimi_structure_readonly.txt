#!/bin/zsh
# AZIMI PUBLISHING ENGINE — READ-ONLY STRUCTURE INSPECTION
# No project files or directories are created, modified, or deleted.
set -u
ROOT="$HOME/Desktop/azimi-publishing-engine"
OUTPUT="$HOME/Desktop/Azimi-Publishing-Output"

printf '\n=== CURRENT DIRECTORY ===\n'
pwd
printf '\n=== PROJECT ROOT ===\n'
if [[ -d "$ROOT" ]]; then
  printf 'FOUND: %s\n' "$ROOT"
  printf '\n=== PROJECT TREE (MAX DEPTH 3; EXCLUDES .venv AND .git) ===\n'
  find "$ROOT" -maxdepth 3 \( -name '.venv' -o -name '.venv-python39-backup-*' -o -name '.git' \) -prune -o -print | sort
  printf '\n=== DIRECTIVE FILES ===\n'
  find "$ROOT" -maxdepth 5 \( -name '.venv' -o -name '.venv-python39-backup-*' -o -name '.git' \) -prune -o -name 'directive.txt' -print
else
  printf 'NOT FOUND: %s\n' "$ROOT"
fi
printf '\n=== PDF OUTPUT DIRECTORY ===\n'
if [[ -d "$OUTPUT" ]]; then
  printf 'FOUND: %s\n' "$OUTPUT"
  ls -la "$OUTPUT"
else
  printf 'NOT FOUND: %s\n' "$OUTPUT"
fi
printf '\n=== INSPECTION COMPLETE (READ ONLY) ===\n'
